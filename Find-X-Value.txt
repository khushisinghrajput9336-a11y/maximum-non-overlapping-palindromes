class Solution {
    public long[] resultArray(int[] nums, int k) {
        long[] dp = new long[k];
        long[] result = new long[k];

        for (int num : nums) {
            long[] next = new long[k];
            int val = num % k;

            next[val]++;

            for (int r = 0; r < k; r++) {
                int nr = (int) ((long) r * val % k);
                next[nr] += dp[r];
            }

            for (int r = 0; r < k; r++) {
                result[r] += next[r];
            }

            dp = next;
        }

        return result;
    }
}
